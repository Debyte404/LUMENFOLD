\documentclass{egpubl}

% --- CGF Journal, submission version.
%
% The class's \JournalSubmission stamp prints "submitted to COMPUTER GRAPHICS
% Forum (<month>/<year>)." Two problems with using it as-is: it asserts a
% submission that has not happened, and the date is \number\month/\number\year,
% i.e. the COMPILE date, so the line silently changes every time the paper is
% rebuilt and two builds of identical source disagree about when it is from.
%
% The statement below is deliberately one that is true before submission and stays
% true after it: it names the venue the paper is prepared for and fixes the date.
% When the paper is actually submitted, change "manuscript prepared for" to
% "submitted to" here and nothing else needs touching.
\makeatletter
\renewcommand\p@journalSubmissionText[1]{%
manuscript prepared for COMPUTER GRAPHICS \textit{Forum}, October 2026.%
}
\makeatother
\JournalSubmission

% The stock template branches on \ifpdf: with pdfTeX it loads graphicx with the
% pdftex driver, otherwise it falls back to dvips, which cannot read PDF figures.
% Tectonic runs XeTeX, where \ifpdf is false, so the dvips branch was taken and
% every \includegraphics{*.pdf} failed. Loading graphicx without a driver option
% lets it detect the engine itself.

% Paragraphs are run-in headings, not numbered sections. The class numbers them by
% default and its \paragraph counter carries an unused subsubsection level, so every
% one of them rendered as "4.1.0.1", "7.8.0.1" -- a visible defect in a journal
% submission. Stopping numbering at the subsection level leaves them as the bold
% run-in heads they are meant to be.
\setcounter{secnumdepth}{2}


\sisetup{detect-all,group-separator={,},group-minimum-digits=4}

% Theorem-like environments. Everything the paper asserts as a formal
% statement is one of these; the remaining prose is measurement.
\theoremstyle{plain}
% theoremcounter is the shared master counter for the environments below. This
% paper states no theorems, so it is never used as an environment; it exists so
% that lemmas, corollaries and propositions share one sequence of numbers.
\newtheorem{theoremcounter}{Theorem}
\newtheorem{lemma}[theoremcounter]{Lemma}
\newtheorem{corollary}[theoremcounter]{Corollary}
\newtheorem{proposition}[theoremcounter]{Proposition}
\theoremstyle{definition}
\newtheorem{definition}[theoremcounter]{Definition}
\theoremstyle{remark}
\newtheorem{remark}[theoremcounter]{Remark}
\newtheorem{assumption}[theoremcounter]{Assumption}
